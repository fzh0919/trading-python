```python
def _is_leverage_exempt_sector(bundle: TickerMarketBundle | None) -> bool:
    if bundle is None:
        return False
    text = f"{bundle.sector or ''} {bundle.industry or ''}".lower()
    return any(kw in text for kw in _LEVERAGE_EXEMPT_SECTOR_KEYWORDS)

def evaluate_isa_value_gate(
    bundle: TickerMarketBundle | None,
    tech: TechnicalSnapshot | None,
    macro: MacroSnapshot | None,
    strategy: StrategySection,
    *,
    item: Any | None = None,
    symbol: str = "",
    crisis_mode: bool = False,
) -> dict[str, Any]:
```

**精确计算逻辑（次级模型逐条照实现，勿简化）**：
1. **数据完全缺失降级**：
   - 若 `bundle is None`：返回 `quality_pass=True`（无法证伪不误杀，标注 `"财务快照缺失待核"`）、`mos_pass=False`（无估值依据不能放行买单，`mos_reasons=["缺少现价/估值数据，无法确认安全边际"]`）、`passed=False`、`bubble_risk=False`。
2. **质量门（Quality Gate，防价值陷阱）**：
   - `quality_fails: list[str] = []`，`quality_pros: list[str] = []`
   - 检查 1（盈利红线）：
     - 若 `bundle.profit_margin_pct is not None and bundle.profit_margin_pct <= 0.0` → 加入 `quality_fails`（`"净利率≤0%(亏损企业)"`）
     - 若 `bundle.roe_pct is not None and bundle.roe_pct <= 0.0` → 加入 `quality_fails`（`"ROE≤0%(资本回报为负)"`）
   - 检查 2（杠杆与流动性红线，**金融/地产/公用事业豁免！**）：
     - 若 `not _is_leverage_exempt_sector(bundle)` 且 `bundle.debt_to_equity is not None`：
       - 若 `bundle.current_ratio is not None`：`de > max_de(180)` **且** `cr < min_cr(0.9)` → 加入 `quality_fails`（`"高杠杆且流动性偏紧(D/E=..., CR=...)"`）
       - 若 `bundle.current_ratio is None`：`de > max_de_no_cr(250)` → 加入 `quality_fails`（`"非金融企业极高杠杆(D/E=...)"`）
   - 检查 3（营收坍塌红线）：
     - 若 `bundle.revenue_growth_pct is not None and bundle.revenue_growth_pct <= min_rev_growth(-10.0)` → 加入 `quality_fails`（`"营收大幅萎缩(<= -10%)"`）
   - 检查 4（现金流失血红线）：
     - 若 `bundle.fcf_yield_pct is not None and bundle.fcf_yield_pct < min_fcf(-2.0)` → 加入 `quality_fails`（`"自由现金流明显为负(FCF率=...%)"`）
   - 质量加分标签（仅展示）：`ROE >= 10%`、`净利率 >= 5%`、`FCF率 >= 3%`。
   - `quality_pass = (len(quality_fails) == 0)`。
3. **估值泡沫判定（`bubble_risk`）**：
   - 取 `pe = bundle.forward_pe if (bundle.forward_pe and bundle.forward_pe > 0) else (bundle.trailing_pe if (bundle.trailing_pe and bundle.trailing_pe > 0) else None)`。
   - `high_pe = (pe is not None and pe >= bubble_pe(45.0))`
   - `growth_insufficient = ((bundle.peg_ratio is not None and bundle.peg_ratio >= bubble_peg(2.8)) or (bundle.revenue_growth_pct is not None and bundle.revenue_growth_pct < 15.0) or (bundle.peg_ratio is None and bundle.revenue_growth_pct is None))`
   - `bubble_risk = bool(high_pe and growth_insufficient)`。
4. **安全边际锚（Margin of Safety）计算**：
   - 确定 `entry`：优先 `float(getattr(item, "entry_price", 0.0) or 0.0)`，若 `<= 0` 回退 `float(bundle.current_price or 0.0)`；做 `_rescale_price_pair(entry, bundle.current_price)` 防 GBp/GBP 100 倍错位。
   - 计算摩擦 `friction_pct = _estimate_entry_friction_pct(symbol or bundle.symbol, bundle, strategy, "ISA", None)`。
   - **锚 A（目标估值折价 `target_mos_pct`）**：
     - 若 `(bundle.analyst_count or 0) >= 3` 且 `bundle.target_mean_price and bundle.target_mean_price > 0`：
       - `ref_target = _rescale_price_pair(float(bundle.target_mean_price), entry)`
       - 若 `item` 给了 `target_price > 0`：取 `llm_t = _rescale_price_pair(float(item.target_price), entry)`，如果 `llm_t <= ref_target * 1.15` 可取 `min(llm_t, ref_target)`（更保守），否则用 `ref_target`。
       - `target_mos_pct = round(((ref_target / entry) - 1.0) * 100.0 - friction_pct, 2)`（其中 `entry > 0`）。
     - 否则若 `item` 给了 `target_price > 0` 且 `entry > 0`：`llm_t = _rescale_price_pair(float(item.target_price), entry)`，但无分析师覆盖时不盲信超高目标价，最多按 `min(((llm_t / entry) - 1.0) * 100.0, 25.0) - friction_pct` 计。
   - **锚 B（盈利收益率溢价 `erp_pct`）**：
     - `rf = float(macro.us_10y_yield) if (macro is not None and macro.us_10y_yield is not None and macro.us_10y_yield > 0) else DEFAULT_RISK_FREE_YIELD_PCT`
     - 若 `pe is not None and pe > 0`：`erp_pct = round((100.0 / pe) - rf, 2)`，否则 `None`。
   - **锚 C（自由现金流收益率 `fcf_yield_pct`）**：直接取 `bundle.fcf_yield_pct`。
   - **锚 D（52 周回撤 `dd52`）**：`tech.drawdown_52w_pct`（若 `tech` 非空）。
5. **安全边际是否达标（`mos_pass`）**：
   - 若 `bubble_risk` 为 True：`mos_pass = False`，加入原因 `"估值处于高PE泡沫区，缺乏安全边际"`。
   - 若 `crisis_mode` 为 False（常规模式）：
     - 条件 1：`target_mos_pct is not None and target_mos_pct >= min_mos(12.0)`
     - 条件 2：`erp_pct is not None and erp_pct >= min_erp(1.0)` **且**（`target_mos_pct is None or target_mos_pct >= 6.0`）**且**（`fcf_yield_pct is None or fcf_yield_pct >= 2.0`）
     - 条件 3（无分析师覆盖但经典低估值）：`target_mos_pct is None` **且** `pe is not None and pe <= 20.0` **且**（`bundle.ev_ebitda is None or bundle.ev_ebitda <= 14.0`）
     - 满足条件 1/2/3 任一 → `mos_pass = True`。
   - 若 `crisis_mode` 为 True（危机日 `VIX >= 28` / `RISK_OFF` 严格深折价模式）：
     - 必须同时满足：
       - `deep_discount = (target_mos_pct is not None and target_mos_pct >= crisis_mos(20.0))`
       - `value_anchor = bool((erp_pct is not None and erp_pct >= crisis_erp(1.5)) or (fcf_yield_pct is not None and fcf_yield_pct >= 4.5) or (dd52 is not None and dd52 <= -15.0))`
       - `mos_pass = bool(deep_discount and value_anchor)`。
6. 返回字典：
   ```python
   return {
       "quality_pass": quality_pass,
       "mos_pass": mos_pass,
       "passed": bool(quality_pass and mos_pass),
   ```

```python
def _is_right_side_leader(tech: TechnicalSnapshot | None) -> bool:
    if tech is None:
        return False
    above_ma200 = (tech.dist_ma200_pct is not None and tech.dist_ma200_pct > 0.0)

def resolve_invest_dynamic_rr_floor(
    regime: str | None,
    tech: TechnicalSnapshot | None,
    strategy: StrategySection,
) -> tuple[float, str]:
    reg = str(regime or "NEUTRAL").strip().upper()
    base_neutral = float(
        getattr(strategy, "invest_min_rr_neutral", None)
        or getattr(strategy, "min_rr_ratio", INVEST_MIN_RR_NEUTRAL)
        or INVEST_MIN_RR_NEUTRAL
    )
    rr_leader = float(getattr(strategy, "invest_min_rr_risk_on_leader", INVEST_MIN_RR_RISK_ON_LEADER))
    rr_weak = float(getattr(strategy, "invest_min_rr_weak_or_risk_off", INVEST_MIN_RR_WEAK_OR_RISK_OFF))

    is_weak = bool(
        reg == "RISK_OFF"
        or (tech is not None and (
            tech.trend_regime == "WEAK"
            or (tech.dist_ma200_pct is not None and tech.dist_ma200_pct <= 0.0)
        ))
    )
    if is_weak:
        return rr_weak, "弱势/防御高赔率门槛"
    if reg == "RISK_ON" and _is_right_side_leader(tech):
        return min(rr_leader, base_neutral), "RISK_ON右侧强结构"
    return base_neutral, f"{reg or 'NEUTRAL'}标准门槛"
```
